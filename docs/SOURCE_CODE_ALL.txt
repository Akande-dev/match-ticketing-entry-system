rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn()   { return request.auth != null; }
    function role()       { return request.auth.token.get('role', 'fan'); }
    function isAdmin()    { return signedIn() && role() == 'admin'; }
    function isManager()  { return signedIn() && (role() == 'manager' || role() == 'admin'); }
    function isOwner(uid) { return signedIn() && request.auth.uid == uid; }
    function validName(n) { return n is string && n.size() > 0 && n.size() <= 100; }

    match /users/{uid} {
      allow read: if isOwner(uid) || isAdmin();
      allow create: if isOwner(uid)
        && request.resource.data.keys().hasOnly(['name', 'email', 'role', 'active', 'createdAt'])
        && request.resource.data.keys().hasAll(['name', 'email', 'role', 'active'])
        && request.resource.data.role == 'fan'
        && request.resource.data.active == true
        && request.resource.data.email == request.auth.token.email
        && validName(request.resource.data.name);
      allow update: if isOwner(uid)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name'])
        && validName(request.resource.data.name);
    }

    match /stadiums/{id}/{document=**} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    match /matches/{matchId} {
      allow read: if resource.data.status == 'OnSale' || isManager();
      allow write: if false;
      match /prices/{category}  { allow read: if signedIn(); allow write: if false; }
      match /inventory/{seatId} { allow read: if signedIn(); allow write: if false; }
    }

    match /orders/{id}    { allow read: if signedIn() && (resource.data.userId == request.auth.uid || isManager());
                            allow write: if false; }
    match /tickets/{id}   { allow read: if signedIn() && (resource.data.userId == request.auth.uid || isManager());
                            allow write: if false; }
    match /refunds/{id}   { allow read: if signedIn() && (resource.data.userId == request.auth.uid || isManager());
                            allow write: if false; }
    match /payments/{id}  { allow read: if isManager(); allow write: if false; }
    match /scanLogs/{id}  { allow read: if isManager(); allow write: if false; }
    match /auditLogs/{id} { allow read: if isAdmin(); allow write: if false; }
    match /stats/{id}     { allow read: if isManager(); allow write: if false; }
    match /rateLimits/{id} { allow read, write: if false; }

    match /{document=**} { allow read, write: if false; }
  }
}

function readEnv() {
  const out = {};
  const file = path.join(__dirname, '..', '.env');
  if (!fs.existsSync(file)) return out;
  fs.readFileSync(file, 'utf8').split('\n').forEach((line) => {
    const m = /^([A-Z0-9_]+)=(.*)$/.exec(line.trim());
    if (m) out[m[1]] = m[2];
  });
  return out;
}

async function commitInChunks(ops) {
  for (let i = 0; i < ops.length; i += 450) {
    const batch = db.batch();
    ops.slice(i, i + 450).forEach((op) => op(batch));
    await batch.commit();
  }
}

  for (const v of VENUES) {
    const venueRef = db.collection('stadiums').doc(v.id);

      const count = s.rows * s.perRow;

      for (let r = 1; r <= s.rows; r += 1) {
        for (let n = 1; n <= s.perRow; n += 1) {
          ops.push((b) => b.set(secRef.collection('seats').doc(`${s.id}-R${r}-S${n}`), { row: String(r), number: n }));
        }
      }
    }

(async () => {
  if (!['admin', 'ticket'].includes(mode) || !email) {
    console.error('Usage: node functions/scripts/user-tools.js <admin|ticket> <email>');
    process.exit(1);
  }
  let user;
  try {
    user = await getAuth().getUserByEmail(email);
  } catch {
    console.error(`No account found for ${email}. Check the spelling, and that the emulators are running.`);
    process.exit(1);
  }

  if (mode === 'admin') {
    await getAuth().setCustomUserClaims(user.uid, { role: 'admin' });
    await getFirestore().collection('users').doc(user.uid).set({ role: 'admin' }, { merge: true });
    console.error(`DONE: ${email} is now an admin.`);
    console.error('Now LOG OUT and LOG BACK IN on the website so it picks up the new role.');
    return;
  }

  const snap = await getFirestore().collection('tickets').where('userId', '==', user.uid).get();
  if (snap.empty) {
    console.error('That account has no tickets yet. Buy one on the website first.');
    process.exit(1);
  }
  const tickets = snap.docs.map((d) => ({ id: d.id, ...d.data() }));
  tickets.forEach((t, i) => console.error(`${i + 1}. ${t.matchLabel} - ${t.seatLabel} - status: ${t.status}`));
  const pick = tickets.find((t) => t.status === 'Valid') ?? tickets[0];
  execSync('clip', { input: pick.qrPayload });
  console.error(`\nCOPIED to clipboard: ${pick.seatLabel} (${pick.status}).`);
  console.error('Now click into the "Manual entry" box on the Scanner page and press Ctrl+V.');
})();

const { DomainError } = require('../shared/errors');
const { isSport } = require('../shared/sports');

const MAX_TEAM_LENGTH = 60;
const MAX_PRICE_MINOR = 100_000_000; // 1,000,000 naira sanity cap

const bad = (msg) => new DomainError('invalid-argument', msg);

function validateMatchInput(input, { categories, nowMs }) {
  const { sport, homeTeam, awayTeam, kickoffMs, stadiumId, prices } = input ?? {};
  if (!isSport(sport)) throw bad('Choose a sport.');

  const home = typeof homeTeam === 'string' ? homeTeam.trim() : '';
  const away = typeof awayTeam === 'string' ? awayTeam.trim() : '';
  if (!home || !away || home.length > MAX_TEAM_LENGTH || away.length > MAX_TEAM_LENGTH) {
    throw bad('Both sides are required (max 60 characters each).');
  }
  if (home.toLowerCase() === away.toLowerCase()) throw bad('Both sides cannot be the same.');
  if (!Number.isFinite(kickoffMs) || kickoffMs <= nowMs) throw bad('Start time must be in the future.');
  if (!prices || typeof prices !== 'object') throw bad('Prices are required.');

  const priceMap = new Map(Object.entries(prices));
  for (const key of priceMap.keys()) {
    if (!categories.includes(key)) throw bad(`Unknown seat category: ${key}.`);
  }
  const clean = new Map();
  for (const category of categories) {
    const amount = priceMap.get(category);
    if (!Number.isInteger(amount) || amount <= 0 || amount > MAX_PRICE_MINOR) {
      throw bad(`Set a valid price for ${category}.`);
    }
    clean.set(category, amount);
  }
  return { sport, homeTeam: home, awayTeam: away, kickoffMs, stadiumId, prices: clean };
}

/** Creates or updates a DRAFT match and its prices. */
async function saveMatchCore(db, { uid, matchId, input, now = new Date() }) {
  const stadiumId = requireString(input?.stadiumId, 'Stadium', 100);
  const sections = await loadSections(db, stadiumId);
  const categories = [...new Set(sections.map((s) => s.data().category))];
  const data = validateMatchInput({ ...input, stadiumId }, { categories, nowMs: now.getTime() });
  const venueSnap = await db.collection('stadiums').doc(stadiumId).get();
  const venueSport = venueSnap.data()?.sport;
  if (venueSport && venueSport !== data.sport) {
    throw new DomainError('failed-precondition', 'That venue is not set up for this sport.');
  }

  const matchRef = matchId
    ? db.collection('matches').doc(requireString(matchId, 'matchId', 100))
    : db.collection('matches').doc();
  if (matchId) {
    const existing = await matchRef.get();
    if (!existing.exists) throw new DomainError('not-found', 'Match not found.');
    if (existing.data().status !== 'Draft') {
      throw new DomainError('failed-precondition', 'Only draft matches can be edited.');
    }
  }

  const batch = db.batch();
  const oldPrices = await matchRef.collection('prices').get();
  oldPrices.docs.forEach((d) => batch.delete(d.ref));
  data.prices.forEach((amountMinor, category) =>
    batch.set(matchRef.collection('prices').doc(category), { amountMinor, currency: 'NGN' }));

  batch.set(matchRef, {
    stadiumId: data.stadiumId, venueName: venueSnap.data()?.name ?? null, sport: data.sport, homeTeam: data.homeTeam, awayTeam: data.awayTeam,
    kickoff: Timestamp.fromMillis(data.kickoffMs), status: 'Draft',
    minPrice: Math.min(...data.prices.values()), updatedAt: FieldValue.serverTimestamp(),
    ...(matchId ? {} : { createdBy: uid, createdAt: FieldValue.serverTimestamp() }),
  }, { merge: true });
  writeAudit(batch, db, { actorId: uid, action: matchId ? 'MATCH_UPDATED' : 'MATCH_CREATED', target: matchRef.id });

  await batch.commit();
  return { matchId: matchRef.id };
}

/** Seeds seat inventory from the stadium layout, then opens sales. Safe to retry. */
async function publishMatchCore(db, { uid, matchId }) {
  const matchRef = db.collection('matches').doc(requireString(matchId, 'matchId', 100));
  const snap = await matchRef.get();
  if (!snap.exists) throw new DomainError('not-found', 'Match not found.');
  const match = snap.data();
  if (match.status !== 'Draft') throw new DomainError('failed-precondition', 'Only draft matches can be published.');
  if (match.kickoff.toMillis() <= Date.now()) throw new DomainError('failed-precondition', 'Kick-off has passed.');

  const priceSnap = await matchRef.collection('prices').get();
  const priced = new Set(priceSnap.docs.map((d) => d.id));
  const sections = await loadSections(db, match.stadiumId);
  if (sections.some((s) => !priced.has(s.data().category))) {
    throw new DomainError('failed-precondition', 'Every seat category needs a price.');
  }

  const ops = [];
  for (const section of sections) {
    const { name, category } = section.data();
    const seats = await section.ref.collection('seats').get();
    seats.forEach((seat) => {
      const { row, number } = seat.data();
      ops.push((b) => b.set(matchRef.collection('inventory').doc(seat.id), {
        sectionId: section.id, category, label: `${name} Row ${row} Seat ${number}`,
        status: 'Available', heldByOrderId: null, holdExpiresAt: null,
      }));
    });
  }
  ops.push((b) => b.set(db.collection('stats').doc(matchRef.id), {
    soldCount: 0, revenueMinor: 0, refundedCount: 0, attendance: 0,
  }, { merge: true }));
  await commitInChunks(db, ops);

  // The status flips last, so a failed run leaves a Draft that can be published again.
  const batch = db.batch();
  batch.update(matchRef, { status: 'OnSale', publishedAt: FieldValue.serverTimestamp() });
  writeAudit(batch, db, { actorId: uid, action: 'MATCH_PUBLISHED', target: matchRef.id });
  await batch.commit();
  return { status: 'OnSale', seats: ops.length - 1 };
}

async function cancelMatchCore(db, { uid, matchId }) {
  const matchRef = db.collection('matches').doc(requireString(matchId, 'matchId', 100));
  await db.runTransaction(async (tx) => {
    const snap = await tx.get(matchRef);
    if (!snap.exists) throw new DomainError('not-found', 'Match not found.');
    if (!['Draft', 'OnSale'].includes(snap.data().status)) {
      throw new DomainError('failed-precondition', 'This match cannot be cancelled.');
    }
    tx.update(matchRef, { status: 'Cancelled', cancelledAt: FieldValue.serverTimestamp() });
    writeAudit(tx, db, { actorId: uid, action: 'MATCH_CANCELLED', target: matchRef.id });
  });
  return { status: 'Cancelled' }; // fans then request refunds
}

module.exports = { saveMatchCore, publishMatchCore, cancelMatchCore };

/** Fallback for a slow or lost webhook: ask the gateway directly, then confirm as the webhook would. */
exports.verifyPayment = callable(null, async ({ auth, data, db }) => {
  const { ref, id, order } = await loadOwnOrder(db, auth, data.orderId);
  if (order.status !== 'Pending' || !order.providerRef) return { status: order.status };

  const result = await createPaymentGateway(process.env).verifyTransaction(order.providerRef);
  if (result.status === 'success' && result.orderId === id) {
    await confirmPaymentCore(db, signer(), {
      orderId: id, providerRef: result.providerRef, amountMinor: result.amountMinor,
    });
  }
  return { status: (await ref.get()).data().status };
}, { rateLimit: { action: 'verify', limit: 10, windowMs: 60_000 } });

exports.paymentWebhook = onRequest(async (req, res) => {
  if (req.method !== 'POST') return res.status(405).send('Method not allowed');
  const gateway = createPaymentGateway(process.env);
  const { valid, event } = gateway.verifyWebhook(req.headers, req.rawBody.toString('utf8'));
  if (!valid) {
    logger.warn('webhook_invalid_signature');
    return res.status(401).send('Invalid signature');
  }
  if (event.status !== 'success') return res.status(200).send('ignored');

  try {
    await confirmPaymentCore(getFirestore(), signer(), {
      orderId: event.orderId, providerRef: event.providerRef, amountMinor: event.amountMinor,
    });
    return res.status(200).send('ok');
  } catch (err) {
    if (err instanceof DomainError) {
      logger.error('webhook_rejected', { orderId: event.orderId, code: err.code, message: err.message });
      return res.status(200).send('rejected'); // permanent error: retrying will not help
    }
    logger.error('webhook_failed', { orderId: event.orderId, message: err.message });
    return res.status(500).send('error'); // transient error: ask the gateway to retry
  }
});

const { FieldValue } = require('firebase-admin/firestore');
const { DomainError } = require('../shared/errors');

/** Records that money is owed back. The id comes from the gateway reference, so replays are harmless. */
function recordRefundDue(tx, db, { providerRef, orderId, order, amountMinor, reason }) {
  tx.set(db.collection('refunds').doc(`pay_${providerRef}`), {
    ticketId: null, orderId, userId: order.userId, matchId: order.matchId,
    amountMinor, reason, status: 'Approved', auto: true, payoutStatus: 'ManualPending',
    createdAt: FieldValue.serverTimestamp(),
  });
  tx.set(db.collection('payments').doc(), {
    orderId, providerRef, status: 'success', amountMinor, createdAt: FieldValue.serverTimestamp(),
  });
  writeAudit(tx, db, { actorId: 'system', action: 'REFUND_DUE', target: orderId, meta: { reason, providerRef } });
}

/** Idempotent. Only ever called from the server (webhook, verify, mock), never from the client. */
async function confirmPaymentCore(db, signer, { orderId, providerRef, amountMinor }) {
  const orderRef = db.collection('orders').doc(orderId);
  const handledRef = db.collection('refunds').doc(`pay_${providerRef}`);

  return db.runTransaction(async (tx) => {
    const [orderSnap, handledSnap] = await Promise.all([tx.get(orderRef), tx.get(handledRef)]);
    if (!orderSnap.exists) throw new DomainError('not-found', 'Order not found.');
    if (handledSnap.exists) return { alreadyHandled: true };

    const order = orderSnap.data();
    const refundCtx = { providerRef, orderId, order, amountMinor };

    if (order.status === 'Paid') {
      if ((order.paidProviderRef ?? order.providerRef) === providerRef) return { alreadyPaid: true };
      recordRefundDue(tx, db, { ...refundCtx, reason: 'Duplicate payment for an order that is already paid' });
      return { refundDue: true };
    }
    if (['Expired', 'Cancelled', 'Failed'].includes(order.status)) {
      recordRefundDue(tx, db, { ...refundCtx, reason: `Payment received for an order that is ${order.status}` });
      return { refundDue: true };
    }
    if (order.status !== 'Pending') {
      throw new DomainError('failed-precondition', `Order is ${order.status}.`);
    }
    if (amountMinor !== order.totalMinor) {
      recordRefundDue(tx, db, { ...refundCtx, reason: 'Amount paid does not match the order total' });
      return { refundDue: true, mismatch: true };
    }

    const matchRef = db.collection('matches').doc(order.matchId);
    const matchSnap = await tx.get(matchRef);
    const seatRefs = order.seatIds.map((id) => matchRef.collection('inventory').doc(id));
    const seatSnaps = await tx.getAll(...seatRefs);
    const matchData = matchSnap.data();

    const holdIntact = matchData.status === 'OnSale' &&
      seatSnaps.every((s) => s.exists && s.data().heldByOrderId === orderId);

    if (!holdIntact) {
      seatSnaps.forEach((s, i) => {
        if (s.exists && s.data().heldByOrderId === orderId) {
          tx.update(seatRefs[i], { status: 'Available', heldByOrderId: null, holdExpiresAt: null });
        }
      });
      tx.update(orderRef, { status: 'Failed', failureReason: 'Could not fulfil after payment' });
      recordRefundDue(tx, db, { ...refundCtx, reason: 'Could not fulfil after payment' });
      return { failed: true, refundDue: true };
    }

    const matchLabel = `${matchData.homeTeam} vs ${matchData.awayTeam}`;
    const priceBySeat = new Map(Object.entries(order.seatPrices ?? {}));
    const counts = new Map();

    seatSnaps.forEach((s, i) => {
      tx.update(seatRefs[i], { status: 'Sold', holdExpiresAt: null });
      const { category, label } = s.data();
      counts.set(category, (counts.get(category) ?? 0) + 1);

      const ticketRef = db.collection('tickets').doc();
      tx.set(ticketRef, {
        orderId, userId: order.userId, matchId: order.matchId, matchLabel,
        kickoff: matchData.kickoff, sport: matchData.sport ?? null, seatId: s.id, seatLabel: label, category,
        priceMinor: priceBySeat.get(s.id) ?? null,
        status: 'Valid', qrPayload: signer.sign(ticketRef.id, order.matchId),
        usedAt: null, usedGate: null, createdAt: FieldValue.serverTimestamp(),
      });
    });

    tx.update(orderRef, {
      status: 'Paid', providerRef, paidProviderRef: providerRef, paidAt: FieldValue.serverTimestamp(),
    });
    tx.set(db.collection('payments').doc(), {
      orderId, providerRef, status: 'success', amountMinor, createdAt: FieldValue.serverTimestamp(),
    });
    tx.set(db.collection('stats').doc(order.matchId), {
      soldCount: FieldValue.increment(order.seatIds.length),
      revenueMinor: FieldValue.increment(order.totalMinor),
      soldByCategory: Object.fromEntries([...counts].map(([c, n]) => [c, FieldValue.increment(n)])),
    }, { merge: true });
    writeAudit(tx, db, {
      actorId: order.userId, action: 'ORDER_PAID', target: orderId, meta: { tickets: order.seatIds.length },
    });

    return { tickets: order.seatIds.length };
  });
}

module.exports = { confirmPaymentCore };

const crypto = require('crypto');
const { PaymentGateway } = require('./PaymentGateway');
const { DomainError } = require('../shared/errors');

class PaystackGateway extends PaymentGateway {
  constructor({ secretKey, appBaseUrl, httpClient = fetch, baseUrl = 'https://api.paystack.co' }) {
    super();
    if (!secretKey) throw new Error('Paystack secret key is required');
    this.secretKey = secretKey;
    this.appBaseUrl = appBaseUrl;
    this.http = httpClient;
    this.baseUrl = baseUrl;
  }

  async createCheckout(order) {
    // A fresh reference per attempt: Paystack rejects a reused reference.
    const reference = `${order.id}-${Date.now().toString(36)}`;
    const res = await this.http(`${this.baseUrl}/transaction/initialize`, {
      method: 'POST',
      headers: { Authorization: `Bearer ${this.secretKey}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({
        email: order.userEmail,
        amount: order.totalMinor, // minor units (kobo)
        currency: 'NGN',
        reference,
        callback_url: `${this.appBaseUrl}/checkout/${order.id}`,
        metadata: { orderId: order.id },
      }),
    });
    const json = await res.json().catch(() => ({}));
    if (!res.ok || !json.status) {
      throw new DomainError('unavailable', 'The payment provider is unavailable. Please try again.');
    }
    return { checkoutUrl: json.data.authorization_url, providerRef: json.data.reference };
  }

  verifyWebhook(headers, rawBody) {
    const expected = crypto.createHmac('sha512', this.secretKey).update(rawBody).digest('hex');
    const given = String(headers['x-paystack-signature'] ?? '');
    const valid = given.length === expected.length &&
      crypto.timingSafeEqual(Buffer.from(given), Buffer.from(expected));
    if (!valid) return { valid: false };

    const body = JSON.parse(rawBody);
    const data = body.data ?? {};
    const succeeded = body.event === 'charge.success' && data.status === 'success' && data.currency === 'NGN';
    return {
      valid: true,
      event: {
        providerRef: data.reference,
        orderId: data.metadata?.orderId,
        status: succeeded ? 'success' : 'ignored',
        amountMinor: data.amount,
      },
    };
  }

  async verifyTransaction(reference) {
    const res = await this.http(`${this.baseUrl}/transaction/verify/${encodeURIComponent(reference)}`, {
      headers: { Authorization: `Bearer ${this.secretKey}` },
    });
    const json = await res.json().catch(() => ({}));
    if (!res.ok || !json.status) {
      throw new DomainError('unavailable', 'The payment provider is unavailable. Please try again.');
    }
    const d = json.data ?? {};
    return {

const REFUND_CUTOFF_MS = 48 * 60 * 60 * 1000; // refunds close 48 hours before kick-off

/** @returns {{ok: boolean, reason: string}} */
function checkRefundEligibility({ ticket, match, existingRefundStatus, nowMs }) {
  if (!ticket) return { ok: false, reason: 'Ticket not found.' };
  if (ticket.status !== 'Valid') return { ok: false, reason: `This ticket is ${ticket.status}.` };
  if (['Requested', 'Approved'].includes(existingRefundStatus)) {
    return { ok: false, reason: 'A refund is already in progress for this ticket.' };
  }
  if (match.status === 'Cancelled') return { ok: true, reason: 'Match cancelled.' };
  if (match.kickoff.toMillis() - nowMs >= REFUND_CUTOFF_MS) return { ok: true, reason: 'Before cut-off.' };
  return { ok: false, reason: 'Refunds close 48 hours before kick-off.' };
}

module.exports = { REFUND_CUTOFF_MS, checkRefundEligibility };

const { FieldValue } = require('firebase-admin/firestore');
const { DomainError } = require('../shared/errors');
const { requireString } = require('../shared/validation');
const { writeAudit } = require('../shared/audit');
const { checkRefundEligibility } = require('./refundLogic');

// The refund document id is the ticket id, so a ticket has at most one live refund record.
async function requestRefundCore(db, { uid, ticketId, reason, now = new Date() }) {
  const id = requireString(ticketId, 'ticketId', 100);
  const cleanReason = typeof reason === 'string' ? reason.trim().slice(0, 200) : '';

  return db.runTransaction(async (tx) => {
    const ticketSnap = await tx.get(db.collection('tickets').doc(id));
    if (!ticketSnap.exists || ticketSnap.data().userId !== uid) {
      throw new DomainError('not-found', 'Ticket not found.');
    }
    const ticket = ticketSnap.data();
    const refundRef = db.collection('refunds').doc(id);
    const [matchSnap, refundSnap] = await Promise.all([
      tx.get(db.collection('matches').doc(ticket.matchId)), tx.get(refundRef),
    ]);

    const verdict = checkRefundEligibility({
      ticket, match: matchSnap.data(), existingRefundStatus: refundSnap.data()?.status, nowMs: now.getTime(),
    });
    if (!verdict.ok) throw new DomainError('failed-precondition', verdict.reason);

    tx.set(refundRef, {
      ticketId: id, orderId: ticket.orderId, userId: uid, matchId: ticket.matchId,
      amountMinor: ticket.priceMinor ?? null, reason: cleanReason,
      status: 'Requested', createdAt: FieldValue.serverTimestamp(),
    });
    writeAudit(tx, db, { actorId: uid, action: 'REFUND_REQUESTED', target: id });
    return { status: 'Requested' };
  });
}

async function decideRefundCore(db, { uid, ticketId, decision, note, now = new Date() }) {
  const id = requireString(ticketId, 'ticketId', 100);
  if (!['approve', 'reject'].includes(decision)) {
    throw new DomainError('invalid-argument', 'Decision must be approve or reject.');
  }
  const cleanNote = typeof note === 'string' ? note.trim().slice(0, 200) : '';

  return db.runTransaction(async (tx) => {
    const refundRef = db.collection('refunds').doc(id);
    const ticketRef = db.collection('tickets').doc(id);
    const [refundSnap, ticketSnap] = await Promise.all([tx.get(refundRef), tx.get(ticketRef)]);
    if (!refundSnap.exists || refundSnap.data().status !== 'Requested' || !ticketSnap.exists) {
      throw new DomainError('failed-precondition', 'No pending refund request for this ticket.');
    }
    const ticket = ticketSnap.data();
    const matchRef = db.collection('matches').doc(ticket.matchId);
    const seatRef = matchRef.collection('inventory').doc(ticket.seatId);
    const [matchSnap, seatSnap] = await Promise.all([tx.get(matchRef), tx.get(seatRef)]);

    const decided = { decidedBy: uid, decidedAt: FieldValue.serverTimestamp(), note: cleanNote };

    if (decision === 'reject') {
      tx.update(refundRef, { status: 'Rejected', ...decided });
      writeAudit(tx, db, { actorId: uid, action: 'REFUND_REJECTED', target: id });
      return { status: 'Rejected' };
    }

    if (ticket.status !== 'Valid') {
      throw new DomainError('failed-precondition', `Ticket is ${ticket.status}; reject this request instead.`);
    }
    tx.update(ticketRef, { status: 'Cancelled', cancelledAt: FieldValue.serverTimestamp() });

    const match = matchSnap.data();
    const seat = seatSnap.data();
    const reusable = match.status === 'OnSale' && match.kickoff.toMillis() > now.getTime() &&
      seat?.status === 'Sold' && seat.heldByOrderId === ticket.orderId;
    if (reusable) {
      tx.update(seatRef, { status: 'Available', heldByOrderId: null, holdExpiresAt: null });
    }

    tx.set(db.collection('stats').doc(ticket.matchId), {
      soldCount: FieldValue.increment(-1),
      revenueMinor: FieldValue.increment(-(ticket.priceMinor ?? 0)),
      refundedCount: FieldValue.increment(1),
      ...(ticket.category ? { soldByCategory: { [ticket.category]: FieldValue.increment(-1) } } : {}),
    }, { merge: true });

    // Returning the money is a manual step for now; we record that it is owed.
    tx.update(refundRef, { status: 'Approved', payoutStatus: 'ManualPending', ...decided });
    writeAudit(tx, db, { actorId: uid, action: 'REFUND_APPROVED', target: id });
    return { status: 'Approved' };
  });
}

module.exports = { requestRefundCore, decideRefundCore };

const { Timestamp } = require('firebase-admin/firestore');

/**
 * Expires one Pending order whose hold has run out and frees its seats.
 * Re-reads the order in a transaction, so a payment landing at the same moment wins the race.
 */
async function expireOrderCore(db, { orderId, now = new Date() }) {
  const orderRef = db.collection('orders').doc(orderId);
  return db.runTransaction(async (tx) => {
    const snap = await tx.get(orderRef);
    const order = snap.data();
    if (!snap.exists || order.status !== 'Pending' || order.holdExpiresAt.toMillis() > now.getTime()) {
      return { expired: false };
    }
    const matchRef = db.collection('matches').doc(order.matchId);
    const seatRefs = order.seatIds.map((id) => matchRef.collection('inventory').doc(id));
    const seatSnaps = await tx.getAll(...seatRefs);
    seatSnaps.forEach((s, i) => {
      if (s.exists && s.data().status === 'Held' && s.data().heldByOrderId === orderId) {
        tx.update(seatRefs[i], { status: 'Available', heldByOrderId: null, holdExpiresAt: null });
      }
    });
    tx.update(orderRef, { status: 'Expired' });
    return { expired: true };
  });
}

async function releaseExpiredHoldsCore(db, { now = new Date(), limit = 200 } = {}) {
  const snap = await db.collection('orders')
    .where('status', '==', 'Pending')
    .where('holdExpiresAt', '<=', Timestamp.fromDate(now))
    .limit(limit)
    .get();
  let expired = 0;
  for (const doc of snap.docs) {
    const result = await expireOrderCore(db, { orderId: doc.id, now });
    if (result.expired) expired += 1;
  }
  return { scanned: snap.size, expired };
}

module.exports = { expireOrderCore, releaseExpiredHoldsCore };

const { DomainError } = require('../shared/errors');

const MAX_SEATS_PER_ORDER = 6;
const MAX_ACTIVE_HOLDS = 2;
const HOLD_MINUTES = 10;

function validateRequest({ matchId, seatIds }) {
  if (typeof matchId !== 'string' || !matchId) {
    throw new DomainError('invalid-argument', 'matchId is required.');
  }
  if (!Array.isArray(seatIds) || seatIds.length === 0) {
    throw new DomainError('invalid-argument', 'Select at least one seat.');
  }
  if (seatIds.length > MAX_SEATS_PER_ORDER) {
    throw new DomainError('invalid-argument', `You can book at most ${MAX_SEATS_PER_ORDER} seats.`);
  }
  if (!seatIds.every((s) => typeof s === 'string' && s) || new Set(seatIds).size !== seatIds.length) {
    throw new DomainError('invalid-argument', 'Seat list is invalid.');
  }
}

/** A seat can be taken if Available, or Held with an expired hold. */
function isSeatAvailable(seat, nowMs) {
  if (!seat) return false;
  if (seat.status === 'Available') return true;
  if (seat.status === 'Held') return (seat.holdExpiresAt?.toMillis?.() ?? 0) <= nowMs;
  return false;
}

function findUnavailable(seatDocs, nowMs) {
  return seatDocs.filter((s) => !isSeatAvailable(s.data, nowMs)).map((s) => s.id);
}

function countActiveHolds(orders, nowMs) {
  return orders.filter(
    (o) => o.status === 'Pending' && (o.holdExpiresAt?.toMillis?.() ?? 0) > nowMs,
  ).length;
}

/** prices: Map<category, amountMinor> */
function computeTotal(seatDocs, prices) {
  return seatDocs.reduce((sum, s) => {
    const price = prices.get(s.data.category);
    if (typeof price !== 'number') {
      throw new DomainError('failed-precondition', `No price set for ${s.data.category}.`);
    }
    return sum + price;
  }, 0);
}

module.exports = {
  MAX_SEATS_PER_ORDER, MAX_ACTIVE_HOLDS, HOLD_MINUTES,
  validateRequest, isSeatAvailable, findUnavailable, countActiveHolds, computeTotal,
};

const { Timestamp, FieldValue } = require('firebase-admin/firestore');
const { DomainError } = require('../shared/errors');
const {
  HOLD_MINUTES, MAX_ACTIVE_HOLDS, validateRequest, findUnavailable, countActiveHolds, computeTotal,
} = require('./reservationLogic');

async function holdSeatsCore(db, { uid, matchId, seatIds, now = new Date() }) {
  validateRequest({ matchId, seatIds });
  const matchRef = db.collection('matches').doc(matchId);
  const orderRef = db.collection('orders').doc();

  return db.runTransaction(async (tx) => {
    // All reads first, then all writes (Firestore transaction rule).
    const matchSnap = await tx.get(matchRef);
    if (!matchSnap.exists || matchSnap.data().status !== 'OnSale') {
      throw new DomainError('failed-precondition', 'This match is not on sale.');
    }

    const pendingSnap = await tx.get(
      db.collection('orders').where('userId', '==', uid).where('status', '==', 'Pending'),
    );
    if (countActiveHolds(pendingSnap.docs.map((d) => d.data()), now.getTime()) >= MAX_ACTIVE_HOLDS) {
      throw new DomainError('resource-exhausted',
        'You already have open seat holds. Complete or cancel them first.');
    }

    const seatRefs = seatIds.map((id) => matchRef.collection('inventory').doc(id));
    const seatSnaps = await tx.getAll(...seatRefs);
    const seatDocs = seatSnaps.map((s) => ({ id: s.id, data: s.exists ? s.data() : undefined }));

    if (findUnavailable(seatDocs, now.getTime()).length) {
      throw new DomainError('failed-precondition', 'Some seats were just taken. Please choose others.');
    }

    const categories = [...new Set(seatDocs.map((s) => s.data.category))];
    const priceSnaps = await tx.getAll(...categories.map((c) => matchRef.collection('prices').doc(c)));
    const prices = new Map(priceSnaps.filter((p) => p.exists).map((p) => [p.id, p.data().amountMinor]));
    const totalMinor = computeTotal(seatDocs, prices);

    const holdExpiresAt = Timestamp.fromMillis(now.getTime() + HOLD_MINUTES * 60 * 1000);
    seatRefs.forEach((ref) =>
      tx.update(ref, { status: 'Held', heldByOrderId: orderRef.id, holdExpiresAt }));
    tx.set(orderRef, {
      userId: uid,
      matchId,
      seatIds,
      seatLabels: seatDocs.map((s) => s.data.label),
      seatPrices: Object.fromEntries(seatDocs.map((s) => [s.id, prices.get(s.data.category)])),
      status: 'Pending',
      totalMinor,
      holdExpiresAt,
      createdAt: FieldValue.serverTimestamp(),
    });

    return { orderId: orderRef.id, totalMinor, holdExpiresAt: holdExpiresAt.toMillis() };
  });
}

/** Cancels a Pending order owned by uid and frees its seats. */
async function cancelOrderCore(db, { uid, orderId }) {
  const orderRef = db.collection('orders').doc(orderId);
  return db.runTransaction(async (tx) => {
    const orderSnap = await tx.get(orderRef);
    if (!orderSnap.exists || orderSnap.data().userId !== uid) {
      throw new DomainError('not-found', 'Order not found.');
    }
    const order = orderSnap.data();
    if (order.status !== 'Pending') return { status: order.status };

    const seatRefs = order.seatIds.map((id) =>
      db.collection('matches').doc(order.matchId).collection('inventory').doc(id));
    const seatSnaps = await tx.getAll(...seatRefs);
    seatSnaps.forEach((s, i) => {
      if (s.exists && s.data().heldByOrderId === orderId) {
        tx.update(seatRefs[i], { status: 'Available', heldByOrderId: null, holdExpiresAt: null });
      }
    });
    tx.update(orderRef, { status: 'Cancelled' });
    return { status: 'Cancelled' };
  });
}

module.exports = { holdSeatsCore, cancelOrderCore };

/** Firestore batches hold at most 500 writes, so large jobs are committed in chunks. */
async function commitInChunks(db, ops, size = 450) {
  for (let i = 0; i < ops.length; i += size) {
    const batch = db.batch();
    ops.slice(i, i + size).forEach((op) => op(batch));
    await batch.commit();
  }
}

/**
 * Wraps a handler with authentication, an optional role check, optional rate limiting
 * and domain-error mapping.
 * @param {string[]|null} roles allowed roles, or null for any signed-in user
 * @param {Function} handler ({ auth, data, db }) => Promise<*>
 * @param {{ rateLimit?: {action, limit, windowMs} }} [options]
 */
function callable(roles, handler, { rateLimit, ...callOptions } = {}) {
  const options = { enforceAppCheck: process.env.ENFORCE_APP_CHECK === 'true', ...callOptions };
  return onCall(options, async (request) => {
    const auth = roles ? requireRole(request, roles) : requireAuth(request);
    try {
      const db = getFirestore();
      if (rateLimit) await enforceRateLimit(db, { uid: auth.uid, ...rateLimit });
      return await handler({ auth, data: request.data ?? {}, db });
    } catch (err) {
      throw toHttpsError(err);
    }
  });
}

const { FieldValue, Timestamp } = require('firebase-admin/firestore');
const { DomainError } = require('./errors');

/** Pure fixed-window counter. */
function nextWindow(state, nowMs, { limit, windowMs }) {
  const start = state?.windowStartMs;
  if (typeof start !== 'number' || nowMs - start >= windowMs) {
    return { allowed: true, state: { windowStartMs: nowMs, count: 1 } };
  }
  if (state.count >= limit) return { allowed: false, retryAfterMs: start + windowMs - nowMs };
  return { allowed: true, state: { windowStartMs: start, count: state.count + 1 } };
}

async function enforceRateLimit(db, { uid, action, limit, windowMs, now = new Date() }) {
  const ref = db.collection('rateLimits').doc(`${uid}_${action}`);
  await db.runTransaction(async (tx) => {
    const snap = await tx.get(ref);
    const verdict = nextWindow(snap.data(), now.getTime(), { limit, windowMs });
    if (!verdict.allowed) {
      const seconds = Math.ceil(verdict.retryAfterMs / 1000);
      throw new DomainError('resource-exhausted', `Too many requests. Try again in ${seconds} seconds.`);
    }
    tx.set(ref, {
      ...verdict.state,
      expireAt: Timestamp.fromMillis(verdict.state.windowStartMs + windowMs * 2),
      updatedAt: FieldValue.serverTimestamp(),
    });
  });
}

module.exports = { nextWindow, enforceRateLimit };

    return snap.exists ? { id: snap.id, ...snap.data() } : null;

  async markUsedIfValid(id, { gate, stewardId, at }) {
    const ref = this.db.collection('tickets').doc(id);
    return this.db.runTransaction(async (tx) => {
      const snap = await tx.get(ref);
      const ticket = { id: snap.id, ...snap.data() };
      if (ticket.status !== 'Valid') return { marked: false, ticket };
      tx.update(ref, { status: 'Used', usedAt: at, usedGate: gate, usedBy: stewardId });
      return { marked: true, ticket };
    });
  }
}

// Each rule has one job: return a failing outcome, or null if the ticket passes.
const ScanResult = Object.freeze({
  VALID: 'Valid',
  ALREADY_USED: 'AlreadyUsed',
  INVALID: 'Invalid',
  WRONG_MATCH: 'WrongMatch',
  CANCELLED: 'Cancelled',
});

class MatchRule {
  check(ticket, ctx) {
    return ticket.matchId === ctx.matchId ? null : { result: ScanResult.WRONG_MATCH };
  }
}

class StatusRule {
  check(ticket) {
    if (ticket.status === 'Valid') return null;
    if (ticket.status === 'Used') {
      return { result: ScanResult.ALREADY_USED, firstUsedAt: ticket.usedAt, firstGate: ticket.usedGate };
    }
    // Cancelled, or any status added in future: refuse. Fail closed, never admit by default.
    return { result: ScanResult.CANCELLED };
  }
}

module.exports = { ScanResult, MatchRule, StatusRule };

const crypto = require('crypto');

/** Signs and verifies QR payloads of the form ticketId.matchId.signature (HMAC-SHA256). */
class TicketSigner {
  constructor(secret) {
    if (!secret) throw new Error('Signing secret is required');
    this.secret = secret;
  }

  _mac(ticketId, matchId) {
    return crypto.createHmac('sha256', this.secret).update(`${ticketId}.${matchId}`).digest('base64url');
  }

  sign(ticketId, matchId) {
    return `${ticketId}.${matchId}.${this._mac(ticketId, matchId)}`;
  }

  /** Returns { ticketId, matchId } or null if malformed or tampered. */
  verify(payload) {
    if (typeof payload !== 'string') return null;
    const first = payload.indexOf('.');
    const last = payload.lastIndexOf('.');
    if (first <= 0 || last <= first + 1 || last === payload.length - 1) return null;

    const ticketId = payload.slice(0, first);
    const matchId = payload.slice(first + 1, last);
    const given = Buffer.from(payload.slice(last + 1));
    const expected = Buffer.from(this._mac(ticketId, matchId));
    if (given.length !== expected.length || !crypto.timingSafeEqual(given, expected)) return null;
    return { ticketId, matchId };
  }
}

module.exports = { TicketSigner };

const { ScanResult } = require('./rules');

/** Orchestrates a gate scan. Depends only on injected collaborators (SOLID: D, I, O). */
class TicketValidator {
  constructor({ signer, tickets, scanLogs, rules, clock }) {
    this.signer = signer;
    this.tickets = tickets;
    this.scanLogs = scanLogs;
    this.rules = rules;
    this.clock = clock;
  }

  async validate(payload, ctx) {
    const outcome = await this._decide(payload, ctx);
    await this.scanLogs.record({
      ticketId: outcome.ticketId ?? null,
      matchId: ctx.matchId,
      gate: ctx.gate,
      stewardId: ctx.stewardId,
      result: outcome.result,
      scannedAt: this.clock.now(),
    });
    return outcome;
  }

  async _decide(payload, ctx) {
    const parsed = this.signer.verify(payload);
    if (!parsed) return { result: ScanResult.INVALID };

    const ticket = await this.tickets.findById(parsed.ticketId);
    if (!ticket) return { result: ScanResult.INVALID };

    for (const rule of this.rules) {
      const failure = rule.check(ticket, ctx);
      if (failure) return { ...failure, ticketId: ticket.id };
    }

    // Atomic step: protects against two gates scanning the same ticket at once.
    const mark = await this.tickets.markUsedIfValid(ticket.id, {
      gate: ctx.gate, stewardId: ctx.stewardId, at: this.clock.now(),
    });
    if (!mark.marked) {
      return {
        result: ScanResult.ALREADY_USED, ticketId: ticket.id,
        firstUsedAt: mark.ticket.usedAt, firstGate: mark.ticket.usedGate,
      };
    }
    return { result: ScanResult.VALID, ticketId: ticket.id, seatLabel: ticket.seatLabel };
  }
}

module.exports = { TicketValidator };

const TRANSFER_CUTOFF_MS = 2 * 60 * 60 * 1000; // transfers close 2 hours before kick-off
const MAX_TRANSFERS = 1; // one transfer per ticket limits resale

/** @returns {{ok: boolean, reason: string}} */
function checkTransferEligibility({ ticket, match, openRefundStatus, nowMs }) {
  if (!ticket) return { ok: false, reason: 'Ticket not found.' };
  if (ticket.status !== 'Valid') return { ok: false, reason: `This ticket is ${ticket.status}.` };
  if ((ticket.transferCount ?? 0) >= MAX_TRANSFERS) {
    return { ok: false, reason: 'This ticket has already been transferred once.' };
  }
  if (['Requested', 'Approved'].includes(openRefundStatus)) {
    return { ok: false, reason: 'Cancel or finish the refund request before transferring.' };
  }
  if (match.status !== 'OnSale') return { ok: false, reason: 'This match is no longer on sale.' };
  if (match.kickoff.toMillis() - nowMs < TRANSFER_CUTOFF_MS) {
    return { ok: false, reason: 'Transfers close 2 hours before kick-off.' };
  }
  return { ok: true, reason: '' };
}

module.exports = { TRANSFER_CUTOFF_MS, MAX_TRANSFERS, checkTransferEligibility };

const { FieldValue } = require('firebase-admin/firestore');
const { DomainError } = require('../shared/errors');
const { requireString } = require('../shared/validation');
const { writeAudit } = require('../shared/audit');
const { checkTransferEligibility } = require('./transferLogic');

const EMAIL_PATTERN = /^[^@\s]+@[^@\s]+\.[^@\s]+$/;

/**
 * Moves a ticket to another registered user by issuing a NEW ticket (new id, new QR)
 * and retiring the old one, so the previous holder's QR code stops working.
 */
async function transferTicketCore(db, signer, { uid, ticketId, recipientEmail, now = new Date() }) {
  const id = requireString(ticketId, 'ticketId', 100);
  const email = requireString(recipientEmail, 'Recipient email', 254).toLowerCase();
  if (!EMAIL_PATTERN.test(email)) throw new DomainError('invalid-argument', 'Enter a valid email address.');

  const found = await db.collection('users').where('email', '==', email).limit(1).get();
  const recipient = found.docs[0];
  // One message for every failure, so nobody can use this form to discover who has an account.
  const cannotTransfer = new DomainError('failed-precondition',
    'We could not transfer to that address. The recipient needs a registered, active account.');
  if (!recipient || recipient.data().active === false) throw cannotTransfer;
  if (recipient.id === uid) throw new DomainError('invalid-argument', 'You already own this ticket.');

  return db.runTransaction(async (tx) => {
    const oldRef = db.collection('tickets').doc(id);
    const oldSnap = await tx.get(oldRef);
    if (!oldSnap.exists || oldSnap.data().userId !== uid) throw new DomainError('not-found', 'Ticket not found.');
    const ticket = oldSnap.data();

    const [matchSnap, refundSnap] = await Promise.all([
      tx.get(db.collection('matches').doc(ticket.matchId)),
      tx.get(db.collection('refunds').doc(id)),
    ]);
    const verdict = checkTransferEligibility({
      ticket, match: matchSnap.data(), openRefundStatus: refundSnap.data()?.status, nowMs: now.getTime(),
    });
    if (!verdict.ok) throw new DomainError('failed-precondition', verdict.reason);

    const newRef = db.collection('tickets').doc();
    tx.set(newRef, {
      ...ticket,
      userId: recipient.id,
      status: 'Valid',
      qrPayload: signer.sign(newRef.id, ticket.matchId),
      usedAt: null,
      usedGate: null,
      transferCount: (ticket.transferCount ?? 0) + 1,
      transferredFrom: id,
      createdAt: FieldValue.serverTimestamp(),
    });
    tx.update(oldRef, { status: 'Transferred', transferredTo: recipient.id, transferredAt: FieldValue.serverTimestamp() });
    writeAudit(tx, db, {
      actorId: uid, action: 'TICKET_TRANSFERRED', target: id, meta: { newTicketId: newRef.id, to: recipient.id },
    });
    return { ticketId: newRef.id };
  });
}

const { randomUUID } = require('crypto');
const { Timestamp } = require('firebase-admin/firestore');

const { confirmPaymentCore } = require('../src/payments/paymentService');
const { TicketSigner } = require('../src/tickets/TicketSigner');

const SECRET = 'test-secret-test-secret-test-secret';
const signer = new TicketSigner(SECRET);
const uniqueId = (prefix) => `${prefix}-${randomUUID().slice(0, 8)}`;

async function seedMatch(db, matchId, seatCount = 2) {
  const m = db.collection('matches').doc(matchId);
  await m.set({
    status: 'OnSale', homeTeam: 'A', awayTeam: 'B', stadiumId: 'st1',
    kickoff: Timestamp.fromMillis(Date.now() + 7 * 24 * 3600000),
  });
  await m.collection('prices').doc('Regular').set({ amountMinor: 200000 });
  for (let i = 1; i <= seatCount; i += 1) {
    await m.collection('inventory').doc(`S${i}`).set({
      status: 'Available', category: 'Regular', label: `Seat ${i}`, heldByOrderId: null, holdExpiresAt: null,
    });
  }
}

/** Holds seats and pays for them, returning the issued tickets. */
async function buyTickets(db, { matchId, uid, seatIds }) {
  const { orderId, totalMinor } = await holdSeatsCore(db, { uid, matchId, seatIds });
  await confirmPaymentCore(db, signer, { orderId, providerRef: `pay-${orderId}`, amountMinor: totalMinor });
  const snap = await db.collection('tickets').where('orderId', '==', orderId).get();
  return { orderId, tickets: snap.docs.map((d) => ({ id: d.id, ...d.data() })) };
}

import { useCallback, useEffect, useState } from 'react';
import { useAuth } from '../auth/AuthContext';
import { listUsers, setUserRole, setUserDisabled } from '../../services/api';

const ROLE_LIST = ['fan', 'steward', 'manager', 'admin'];

export default function UsersPage() {
  const { user } = useAuth();
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [busy, setBusy] = useState('');

  const load = useCallback(async () => {
    try {
      const res = await listUsers({});
      setUsers(res.users);
      setError('');
    } catch (e) {
      setError(e.message || 'Could not load users.');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => { load(); }, [load]);

  async function run(uid, fn) {
    setBusy(uid);
    setError('');
    try { await fn(); await load(); } catch (e) { setError(e.message || 'Action failed.'); }
    setBusy('');
  }

  if (loading) return <p>Loading users...</p>;

  return (
    <section>
      <h2>Users</h2>
      <p>Role changes and disabling are checked and logged by the server. Your own row is read-only. Users must log in again for a new role to apply.</p>
      {error && <p role="alert">{error}</p>}
      <table>
        <thead><tr><th>Name</th><th>Email</th><th>Role</th><th>Status</th><th>Action</th></tr></thead>
        <tbody>
          {users.map((u) => {
            const self = u.uid === user?.uid;
            return (
              <tr key={u.uid}>
                <td>{u.name || '-'}</td>
                <td>{u.email}</td>
                <td>
                  <select
                    aria-label={'Role for ' + u.email}
                    value={u.role}
                    disabled={self || busy === u.uid}
                    onChange={(e) => run(u.uid, () => setUserRole({ uid: u.uid, role: e.target.value }))}
                  >
                    {ROLE_LIST.map((r) => <option key={r} value={r}>{r}</option>)}
                  </select>
                </td>
                <td>{u.disabled ? 'Disabled' : 'Active'}</td>
                <td>
                  <button
                    type="button"
                    className="btn"
                    disabled={self || busy === u.uid}
                    onClick={() => run(u.uid, () => setUserDisabled({ uid: u.uid, disabled: !u.disabled }))}
                  >
                    {u.disabled ? 'Enable' : 'Disable'}
                  </button>
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </section>
  );
}

import { createContext, useContext, useEffect, useMemo, useState } from 'react';
import {
  createUserWithEmailAndPassword, onAuthStateChanged, signInWithEmailAndPassword, signOut, updateProfile,
} from 'firebase/auth';
import { doc, serverTimestamp, setDoc } from 'firebase/firestore';
import { auth, db } from '../../services/firebase';
import { normaliseRole, ROLES } from './roles';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [role, setRole] = useState(ROLES.FAN);
  const [loading, setLoading] = useState(true);

  useEffect(() => onAuthStateChanged(auth, async (u) => {
    if (u) {
      const token = await u.getIdTokenResult(true);
      setRole(normaliseRole(token.claims.role));
    } else {
      setRole(ROLES.FAN);
    }
    setUser(u);
    setLoading(false);
  }), []);

  const value = useMemo(() => ({
    user, role, loading,
    login: (email, password) => signInWithEmailAndPassword(auth, email, password),
    async register(name, email, password) {
      const cred = await createUserWithEmailAndPassword(auth, email, password);
      await updateProfile(cred.user, { displayName: name });
      await setDoc(doc(db, 'users', cred.user.uid), {
        name, email: cred.user.email, role: ROLES.FAN, active: true, createdAt: serverTimestamp(),
      });
    },
    logout: () => signOut(auth),
  }), [user, role, loading]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth must be used inside <AuthProvider>');
  return ctx;
}

export default function ProtectedRoute({ allow, children }) {
  const { user, role, loading } = useAuth();
  const location = useLocation();
  if (loading) return <p>Loadingâ€¦</p>;

export const ROLES = Object.freeze({ FAN: 'fan', STEWARD: 'steward', MANAGER: 'manager', ADMIN: 'admin' });

export function normaliseRole(claimRole) {
  return Object.values(ROLES).includes(claimRole) ? claimRole : ROLES.FAN;
}
export function hasAnyRole(role, allowed) {
  return allowed.includes(normaliseRole(role));
}

import { useEffect, useState } from 'react';
import { useNavigate, useParams, useSearchParams } from 'react-router-dom';
import { doc, onSnapshot } from 'firebase/firestore';
import { db } from '../../services/firebase';
import { cancelOrder, createCheckout, verifyPayment } from '../../services/api';
import { formatMoney } from '../../utils/money';

const clock = (ms) => {
  const s = Math.ceil(ms / 1000);
  return `${String(Math.floor(s / 60)).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;
};

export default function CheckoutPage() {
  const { orderId } = useParams();
  const navigate = useNavigate();
  const [params] = useSearchParams();
  const [order, setOrder] = useState(null);
  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);
  const [now, setNow] = useState(Date.now());
  const returned = params.has('reference') || params.has('trxref');

  useEffect(() => onSnapshot(doc(db, 'orders', orderId),
    (s) => setOrder(s.exists() ? { id: s.id, ...s.data() } : null), () => setError('Order not found.')), [orderId]);
  useEffect(() => { const t = setInterval(() => setNow(Date.now()), 1000); return () => clearInterval(t); }, []);
  useEffect(() => { if (order?.status === 'Paid') navigate('/my-tickets', { replace: true }); }, [order, navigate]);
  useEffect(() => { if (returned) verifyPayment({ orderId }).catch(() => {}); }, [orderId, returned]);

  if (error) return <main><p role="alert">{error}</p></main>;
  if (!order) return <main><p>Loadingâ€¦</p></main>;
  if (order.status !== 'Pending') return <main><p>This order is {order.status}. <a href="/">Back to matches</a></p></main>;

  const remaining = Math.max(0, order.holdExpiresAt.toMillis() - now);

  async function handlePay() {
    setBusy(true);
    setError('');
    try {
      const { checkoutUrl } = await createCheckout({ orderId });
      if (checkoutUrl.startsWith('/')) navigate(checkoutUrl);
      else window.location.assign(checkoutUrl);
    } catch (err) {
      setError(err.message || 'Could not start payment.');
      setBusy(false);
    }
  }
  async function handleCancel() {
    await cancelOrder({ orderId });
    navigate('/', { replace: true });
  }

  return (
    <main>
      <h1>Checkout</h1>
      <p aria-live="polite">{remaining === 0 ? 'Your hold has expired.' : `Seats held for ${clock(remaining)}`}</p>
      {returned && <p>Confirming your paymentâ€¦</p>}
      <ul>{order.seatLabels.map((l) => <li key={l}>{l}</li>)}</ul>
      <p><strong>Total: {formatMoney(order.totalMinor)}</strong></p>
      {error && <p role="alert">{error}</p>}
      <button type="button" onClick={handlePay} disabled={busy || remaining === 0}>Pay now</button>{' '}
      <button type="button" onClick={handleCancel} disabled={busy}>Cancel and release seats</button>
    </main>
  );
}

import { GROUPS, SPORTS, sportsInGroup } from '../../utils/sports';

const EMPTY = { sport: 'football', stadiumId: '', homeTeam: '', awayTeam: '', kickoff: '', prices: {} };
const toLocalInput = (ms) => new Date(ms - new Date(ms).getTimezoneOffset() * 60000).toISOString().slice(0, 16);

export default function MatchFormPage() {
  const { matchId } = useParams();
  const navigate = useNavigate();
  const [stadiums, setStadiums] = useState([]);
  const [categories, setCategories] = useState([]);
  const [form, setForm] = useState(EMPTY);
  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    getDocs(collection(db, 'stadiums')).then((s) => setStadiums(s.docs.map((d) => ({ id: d.id, ...d.data() }))));
  }, []);

  useEffect(() => {
    if (!form.stadiumId) { setCategories([]); return; }
    getDocs(collection(db, 'stadiums', form.stadiumId, 'sections')).then((s) =>
      setCategories([...new Set(s.docs.map((d) => d.data().category))]));
  }, [form.stadiumId]);

  useEffect(() => {
    if (!matchId) return;
    (async () => {
      const [m, p] = await Promise.all([getDoc(doc(db, 'matches', matchId)), getDocs(collection(db, 'matches', matchId, 'prices'))]);
      if (!m.exists()) { setError('Match not found.'); return; }
      const prices = {};
      p.forEach((d) => { prices[d.id] = String(d.data().amountMinor / 100); });
      const data = m.data();
      setForm({ sport: data.sport ?? 'football', stadiumId: data.stadiumId, homeTeam: data.homeTeam,
        awayTeam: data.awayTeam, kickoff: toLocalInput(data.kickoff.toMillis()), prices });

    } catch (err) {
      setError(err.message || 'Could not save the match.');
      setBusy(false);
    }
  }

  return (
    <section>
      <h2>{matchId ? 'Edit draft match' : 'New match'}</h2>
      <form onSubmit={handleSubmit}>
        <label htmlFor="sport">Sport</label>
        <select id="sport" value={form.sport} onChange={changeSport} disabled={Boolean(matchId)}>

            <optgroup key={g} label={g}>
              {sportsInGroup(g).map((id) => <option key={id} value={id}>{SPORTS[id].label}</option>)}
            </optgroup>
          ))}
        </select>
        <label htmlFor="stadium">Venue</label>
        <select id="stadium" value={form.stadiumId} onChange={setField('stadiumId')} required>
          <option value="">Selectâ€¦</option>
          {venues.map((s) => <option key={s.id} value={s.id}>{s.name}</option>)}
        </select>
        <label htmlFor="home">{sides[0]}</label>
        <input id="home" value={form.homeTeam} onChange={setField('homeTeam')} maxLength={60} required />
        <label htmlFor="away">{sides[1]}</label>
        <input id="away" value={form.awayTeam} onChange={setField('awayTeam')} maxLength={60} required />
        <label htmlFor="kickoff">Start time</label>
        <input id="kickoff" type="datetime-local" value={form.kickoff} onChange={setField('kickoff')} required />
        {categories.map((c) => (
          <div key={c}>
            <label htmlFor={`price-${c}`}>{c} price (â‚¦)</label>
            <input id={`price-${c}`} type="number" min="1" step="1" value={form.prices[c] ?? ''} onChange={setPrice(c)} required />
          </div>
        ))}
        {error && <p role="alert">{error}</p>}
        <button type="submit" disabled={busy || !form.stadiumId}>{busy ? 'Savingâ€¦' : 'Save draft'}</button>

    }
  }

  if (tickets.loading) return <main><p>Loading ticketsâ€¦</p></main>;
  if (!tickets.docs.length) return <main><p>You have no tickets yet.</p></main>;

  return (
    <main>
      <h1>My tickets</h1>
      {error && <p role="alert">{error}</p>}
      {notice && <p role="status">{notice}</p>}
      <ul className="cards">
        {tickets.docs.map((t) => {
          const refund = refundByTicket.get(t.id);
          const refundOpen = refund && refund.status !== 'Rejected';
          const canAct = t.status === 'Valid' && !refundOpen;
          const canTransfer = canAct && (t.transferCount ?? 0) < 1;
          return (
            <li key={t.id} className="card">
              <h2>{sportEmoji(t.sport)} {t.matchLabel}</h2>
              <p>{t.seatLabel}</p>
              {t.status === 'Valid' && <QRCodeSVG value={t.qrPayload} size={180} />}
              <p>Status: <strong>{t.status.toUpperCase()}</strong></p>
              {t.transferredFrom && <p>Received by transfer.</p>}
              {t.status === 'Transferred' && <p>You transferred this ticket to another fan.</p>}
              {refund && <p>{REFUND_LABEL[refund.status]}{refund.note ? `: ${refund.note}` : ''}</p>}

              {canAct && transferId !== t.id && (
                <p>
                  <button type="button" onClick={() => handleRefund(t.id)}>Request refund</button>{' '}
                  {canTransfer && (
                    <button type="button" onClick={() => { setTransferId(t.id); setError(''); setNotice(''); }}>
                      Transfer to a friend
                    </button>
                  )}
                </p>
              )}

              {transferId === t.id && (
                <form onSubmit={handleTransfer}>
                  <label htmlFor={`to-${t.id}`}>Friend's email (they must have an account)</label>
                  <input id={`to-${t.id}`} type="email" value={email} onChange={(e) => setEmail(e.target.value)} required />
                  <p>Your copy will stop working as soon as you send.</p>
                  <button type="submit" disabled={busy}>{busy ? 'Sendingâ€¦' : 'Send ticket'}</button>{' '}
                  <button type="button" disabled={busy} onClick={() => setTransferId('')}>Cancel</button>
                </form>
              )}
            </li>
          );
        })}
      </ul>
    </main>
  );
}

import { useEffect, useState } from 'react';
import { onSnapshot } from 'firebase/firestore';

/** `makeQuery` returns a Firestore query, or a falsy value to stay idle. */
export function useLiveQuery(makeQuery, deps) {
  const [state, setState] = useState({ docs: [], loading: true, error: '' });
  useEffect(() => {
    const q = makeQuery();
    if (!q) return undefined;
    return onSnapshot(
      q,
      (snap) => setState({ docs: snap.docs.map((d) => ({ id: d.id, ...d.data() })), loading: false, error: '' }),
      () => setState({ docs: [], loading: false, error: 'Could not load data.' }),
    );
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps);
  return state;
}

// Keep in step with functions/src/shared/sports.js. Emoji are written as escapes so they survive any editor.
export const SPORTS = {
  football: { label: 'Football', group: 'Team ball sports', emoji: '\u26BD', sides: ['Home team', 'Away team'] },
  basketball: { label: 'Basketball', group: 'Team ball sports', emoji: '\u{1F3C0}', sides: ['Home team', 'Away team'] },
  volleyball: { label: 'Volleyball', group: 'Team ball sports', emoji: '\u{1F3D0}', sides: ['Home team', 'Away team'] },
  'table-tennis': { label: 'Table tennis', group: 'Racket sports', emoji: '\u{1F3D3}', sides: ['Player 1', 'Player 2'] },
  boxing: { label: 'Boxing', group: 'Combat sports', emoji: '\u{1F94A}', sides: ['Fighter 1', 'Fighter 2'] },
};

export const SPORT_IDS = Object.keys(SPORTS);
export const GROUPS = [...new Set(Object.values(SPORTS).map((s) => s.group))];
export const sportsInGroup = (group) => SPORT_IDS.filter((id) => SPORTS[id].group === group);

// Matches created before sports existed have no sport field; treat them as football.
const norm = (id) => (Object.prototype.hasOwnProperty.call(SPORTS, id) ? id : 'football');
export const sportOf = (match) => norm(match?.sport);
export const sportLabel = (id) => SPORTS[norm(id)].label;
export const sportEmoji = (id) => SPORTS[norm(id)].emoji;
